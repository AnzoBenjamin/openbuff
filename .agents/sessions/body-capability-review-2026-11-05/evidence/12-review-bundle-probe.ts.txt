import fs from 'node:fs'
import path from 'node:path'
import { spawnSync } from 'node:child_process'
import {
  getChangeReviewBundle,
  computeWorktreeSnapshotIdentity,
} from '/home/ben/Code/CLI/openbuff/sdk/src/tools/get-change-review-bundle.ts'
import { runTargetedValidation } from '/home/ben/Code/CLI/openbuff/sdk/src/tools/run-targeted-validation.ts'
import { LocalHarnessStore } from '/home/ben/Code/CLI/openbuff/sdk/src/services/local-harness-store.ts'

// Git processes used by the real SDK inherit these public test-only settings.
// This prevents fixture setup from consulting private global/system Git config.
process.env.GIT_CONFIG_NOSYSTEM = '1'
process.env.GIT_CONFIG_GLOBAL = '/dev/null'
process.env.GIT_CONFIG_COUNT = '0'
const evidenceDir = '/tmp/openbuff-body-review-2026-11-05/evidence'
const fixtureRoot = fs.mkdtempSync(path.join(evidenceDir, '12-review-bundle-fixture-'))
const stateDir = path.join(fixtureRoot, 'state')
const cwd = path.join(fixtureRoot, 'repo')
fs.mkdirSync(cwd)
const git = (...args: string[]) => {
  const result = spawnSync('git', args, { cwd, encoding: 'utf8' })
  if (result.status !== 0) throw new Error('fixture git failed: ' + args.join(' ') + ': ' + result.stderr)
}
git('-c', 'init.templateDir=', 'init')
git('config', 'user.email', 'probe@example.invalid')
git('config', 'user.name', 'Openbuff audit probe')
git('config', 'core.hooksPath', '/dev/null')
fs.writeFileSync(path.join(cwd, 'README.md'), '# Test fixture\n')
git('add', 'README.md')
git('commit', '-m', 'fixture base')

const prefix = '/*' + 'x'.repeat(1_200_000) + '*/\n'
const bigSourcePath = path.join(cwd, 'untracked.ts')
fs.writeFileSync(bigSourcePath, prefix + 'export const value = 0\n')
const partValue = (parts: unknown[]) => (parts[0] as { value: Record<string, unknown> }).value
const beforeBundle = partValue(await getChangeReviewBundle({ cwd, stateDir, skipSecurityScan: true }))
if (typeof beforeBundle.snapshotId !== 'string') throw new Error('bundle did not return a snapshot')
const beforeHash = beforeBundle.snapshotId
const beforeSize = fs.statSync(bigSourcePath).size
fs.writeFileSync(bigSourcePath, prefix + 'export const value = 1\n')
const afterBundle = partValue(await getChangeReviewBundle({ cwd, stateDir, skipSecurityScan: true }))
const afterSize = fs.statSync(bigSourcePath).size
const identityBeforeValidation = await computeWorktreeSnapshotIdentity({ cwd })
if ('errorMessage' in identityBeforeValidation) throw new Error(identityBeforeValidation.errorMessage)
const targeted = partValue(await runTargetedValidation({
  cwd,
  snapshotId: identityBeforeValidation.snapshotId,
  files: ['untracked.ts'],
  runHooks: async () => {
    fs.writeFileSync(bigSourcePath, prefix + 'export const value = 2\n')
    return [{ type: 'json', value: [{ exitCode: 0, stdout: '', stderr: '', hookName: 'fixture validation' }] }] as any
  },
}))

const store = new LocalHarnessStore(stateDir)
const timestamp = new Date().toISOString()
store.put('findings', {
  schemaVersion: 1, id: 'open-finding', revision: 0,
  repositoryId: beforeBundle.repositoryId as string,
  workspaceId: beforeBundle.workspaceId as string,
  runId: 'fixture-run', snapshotId: beforeHash,
  createdAt: timestamp, updatedAt: timestamp,
  files: ['untracked.ts'], status: 'open', text: 'fixture blocker',
})
const validFindingBundle = partValue(await getChangeReviewBundle({ cwd, stateDir, skipSecurityScan: true }))
const findingFile = path.join(stateDir, beforeBundle.repositoryId as string, 'findings', 'open-finding.json')
fs.writeFileSync(findingFile, '{broken-json')
const corruptFindingBundle = partValue(await getChangeReviewBundle({ cwd, stateDir, skipSecurityScan: true }))
const quarantinedFiles = fs.readdirSync(path.join(path.dirname(findingFile), '.corrupt'))
const result = {
  method: 'Real public review-bundle, snapshot-identity and targeted-validation functions over a local Git fixture; hook runner injected; security scan skipped; no model/network or private configuration reads.',
  fixtureRoot,
  equalLengthUntrackedTail: {
    sourceBytes: beforeSize,
    sameSize: beforeSize === afterSize,
    beforeSnapshot: beforeHash,
    afterSnapshot: afterBundle.snapshotId,
    snapshotChanged: beforeHash !== afterBundle.snapshotId,
    files: beforeBundle.files,
    bundleTruncated: beforeBundle.truncated,
    tailBefore: 'export const value = 0', tailAfter: 'export const value = 1',
  },
  midValidationTailMutation: {
    sourceTailAfterHook: 'export const value = 2',
    sourceMutatedInsideValidation: true,
    result: targeted,
  },
  corruptFindingConsumption: {
    beforeFindings: validFindingBundle.findings,
    afterFindings: corruptFindingBundle.findings,
    returnedError: corruptFindingBundle.errorMessage ?? null,
    diagnosticsReturned: Object.keys(corruptFindingBundle).filter(key => /diagnostic|corrupt|quarantin/i.test(key)),
    quarantinedFiles,
  },
  limits: ['The equal-length tail example targets the review/validation snapshot implementation, not the already-recorded native audit inventory hash probe.', 'No actual target-project tests or model review ran; the injected validation hook demonstrates the source identity gate behavior.'],
}
fs.writeFileSync(path.join(evidenceDir, '12-review-bundle-probe.json'), JSON.stringify(result, null, 2) + '\n')
console.log(JSON.stringify({ tail: result.equalLengthUntrackedTail, validation: result.midValidationTailMutation.result, corruptFinding: result.corruptFindingConsumption }, null, 2))
