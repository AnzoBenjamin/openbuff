function hookHarness(validation = async () => ({ success: true, errors: [] })) {
  const chain = { current: false }
  let canQueue = true
  const statuses: string[] = []
  const hook = useSendMessage({
    inputRef: { current: null },
    activeSubagentsRef: { current: new Set() },
    isChainInProgressRef: chain,
    setStreamStatus: (status: string) => statuses.push(status),
    setContextWindowUsage: noop,
    setCompactionNotice: noop,
    setCanProcessQueue: (value: boolean) => { canQueue = value },
    abortControllerRef: { current: null },
    onBeforeMessageSend: validation,
    mainAgentTimer: {
      start: noop, stop: noop, pause: noop, resume: noop,
      elapsedSeconds: 0, startTime: null, isPaused: false,
    },
    scrollToLatest: noop,
    continueChat: false,
  } as any)
  return { hook, chain, statuses, get canQueue() { return canQueue } }
}

function routeHarness(content: string) {
  let prepared: any
  const params = {
    inputValue: content, agentMode: 'DEFAULT', isStreaming: false,
    isChainInProgressRef: { current: false }, streamMessageIdRef: { current: null },
    abortControllerRef: { current: null }, inputRef: { current: null },
    sendMessage: async (request: any) => {
      prepared = await prepareUserMessage({
        content: request.content, agentMode: request.agentMode,
        deps: {
          setMessages: useChatStore.getState().setMessages,
          lastMessageMode: null, setLastMessageMode: noop,
          scrollToLatest: noop, setHasReceivedPlanResponse: noop,
        },
      })
    },
    addToQueue: () => { throw new Error('Unexpected queue in idle probe') },
    setMessages: useChatStore.getState().setMessages,
    setInputValue: noop, saveToHistory: noop, setInputFocused: noop,
    setCanProcessQueue: noop, scrollToLatest: noop, clearMessages: noop, stopStreaming: noop,
  }
  return { params, get prepared() { return prepared } }
}

test('observed: attach with a ready stub server still fails the local provider gate', async () => {
  clientModule.setAttachTarget({ socketPath: '/tmp/01-stub-server.sock', token: 'fixture' })
  let remoteRuns = 0
  spyOn(clientModule, 'getCodebuffClient').mockImplementation(async () => ({
    backend: { run: async () => { remoteRuns++; return { output: { type: 'success' } } } },
    run: async () => { remoteRuns++; return { output: { type: 'success' } } },
  } as any))
  const harness = hookHarness()
  await harness.hook.sendMessage({ content: 'inspect project', agentMode: 'DEFAULT' })
  const errors = useChatStore.getState().messages.map((m: any) => m.userError).filter(Boolean)
  expect(remoteRuns).toBe(0)
  expect(errors.join(' ')).toContain('provider is not configured')
  expect(harness.chain.current).toBe(false)
  observations.push({ probe: 'attach-local-readiness', remoteRuns, errors, chainBusy: harness.chain.current })
})

test('observed: prompt-producing slash command consumes a still-processing file', async () => {
  useChatStore.getState().addPendingAttachment({
    kind: 'file', id: 'loading-file', path: 'src/large.ts', filename: 'large.ts',
    isDirectory: false, content: '', status: 'processing',
  } as any)
  const ordinary = routeHarness('inspect project')
  await routeUserPrompt(ordinary.params as any)
  expect(ordinary.prepared).toBeUndefined()
  expect(useChatStore.getState().pendingAttachments).toHaveLength(1)
  const command = routeHarness('/mode:default inspect project')
  await routeUserPrompt(command.params as any)
  expect(command.prepared.finalContent).toBe('inspect project')
  expect(useChatStore.getState().pendingAttachments).toEqual([])
  const user = useChatStore.getState().messages.find((m: any) => m.variant === 'user')
  expect(user?.fileAttachments?.length ?? 0).toBe(0)
  observations.push({ probe: 'slash-processing-file', finalContent: command.prepared.finalContent, pendingCount: 0, storedFileCount: user?.fileAttachments?.length ?? 0 })
})

test('observed: plan artifact resolver and reader follow a session symlink outside the project', async () => {
  const fixture = '/tmp/openbuff-body-review-2026-11-05/evidence/01-plan-fixture'
  const project = `${fixture}/project`
  const outside = `${fixture}/outside`
  mkdirSync(`${project}/.agents/sessions`, { recursive: true })
  mkdirSync(outside, { recursive: true })
  writeFileSync(`${outside}/PLAN.md`, 'Outside project fixture content; no credentials.')
  const link = `${project}/.agents/sessions/link`
  if (!existsSync(link)) symlinkSync(outside, link, 'dir')
  setProjectRoot(project)
  const resolved = planArtifacts.resolvePlanSessionDir('link')
  const artifacts = planArtifacts.readPlanArtifacts('link')
  const page = await planArtifacts.readPlanArtifactPage('link', 'PLAN.md')
  expect(resolved.ok).toBe(true)
  expect(realpathSync(link).startsWith(project)).toBe(false)
  expect(artifacts?.files['PLAN.md']).toContain('Outside project fixture content')
  expect(page).toContain('Outside project fixture content')
  observations.push({ probe: 'plan-session-symlink', lexicalAccepted: resolved.ok, physicalInsideProject: false, prefixReaderIncludedOutsideBytes: true, pageReaderIncludedOutsideBytes: true })
})
