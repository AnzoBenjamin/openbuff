const pidPath = path.join(fixtures, '11-child.pid')
const completedPath = path.join(fixtures, '11-child-after-abort.txt')
const childFile = path.join(fixtures, '11-cancel-child.cjs')
if (fs.existsSync(completedPath)) fs.unlinkSync(completedPath)
fs.writeFileSync(childFile, `const fs = require('node:fs'); fs.writeFileSync(${JSON.stringify(pidPath)}, String(process.pid)); setTimeout(() => fs.writeFileSync(${JSON.stringify(completedPath)}, 'fixture work continued'), 300); setInterval(() => {}, 10000);\n`)
const controller = new AbortController()
const pending = runFinalCheckCommands([{ id: 'orphan-check', command: `/usr/bin/node ${childFile} & wait`, timeoutMs: 5000 }], fixtures, undefined, controller.signal)
let childPid: number | undefined
try {
  const deadline = Date.now() + 3000
  while (!fs.existsSync(pidPath) && Date.now() < deadline) await new Promise((resolve) => setTimeout(resolve, 20))
  if (!fs.existsSync(pidPath)) throw new Error('Fixture child never wrote PID')
  childPid = Number(fs.readFileSync(pidPath, 'utf8'))
  controller.abort(new Error('Injected fixture cancellation'))
  const [output] = await pending
  let stillAlive = false
  try { process.kill(childPid, 0); stillAlive = true } catch {}
  await new Promise((resolve) => setTimeout(resolve, 400))
  observations.finalCheckCancellation = { outcome: output.outcome, exitCode: output.exitCode, childAliveWhenCheckSettled: stillAlive, childProducedWorkAfterCancellation: fs.existsSync(completedPath), childPid, fixtureChildKilledInFinally: true }
} finally {
  controller.abort()
  if (childPid) { try { process.kill(childPid, 'SIGKILL') } catch {} }
}

const serialized = JSON.stringify(observations, (_key, value) => typeof value === 'number' && !Number.isFinite(value) ? String(value) : value, 2)
