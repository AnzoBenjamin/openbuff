import json, pathlib, hashlib, re

base = pathlib.Path('/tmp/openbuff-body-review-2026-11-05')
inventory = json.loads((base / 'INVENTORY.json').read_text())
snapshot = inventory['snapshotId']
all_reads = json.loads((base / 'evidence/10-read-ranges.json').read_text())
inventory_paths = set(inventory['files'])
files = sorted({r['path'] for r in all_reads if r['path'] in inventory_paths})
domains = ['security','correctness','state-mutation','error-handling','performance','dependency-hygiene','test-coverage','api-contract']
ranges = []
for file in files:
    selected = sorted((r['from'],r['to']) for r in all_reads if r['path'] == file)
    for first,last in selected:
        if ranges and ranges[-1]['path'] == file and first <= ranges[-1]['to'] + 1:
            ranges[-1]['to'] = max(last,ranges[-1]['to'])
        else:
            ranges.append({'path':file,'from':first,'to':last})
