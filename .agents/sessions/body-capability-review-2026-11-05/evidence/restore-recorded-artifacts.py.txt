"""Restore captured evidence artifacts as data; no recorded command is executed."""
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
from pathlib import Path
import re
import shlex

SPEC = importlib.util.spec_from_file_location('recorded_review_data', Path(__file__).with_name('recover-recorded-data.py'))
DATA = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(DATA)
BASE, EVIDENCE, RECOVERY = DATA.BASE, DATA.EVIDENCE, DATA.RECOVERY
META = {entry['capture']: entry for entry in DATA.INDEX}
ARTIFACTS = []


def save(name, raw, provenance, completeness, transformation):
    path = EVIDENCE / name
    assert path.parent == EVIDENCE
    path.write_bytes(raw)
    record = {'path': 'evidence/' + name, 'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw), 'provenance': provenance, 'completeness': completeness, 'transformation': transformation, 'observationMeaning': 'Recorded baseline evidence; diagnostic successes reproduce defects and do not certify a repair.'}
    ARTIFACTS.append(record)
    return record


def captured_json(name, capture):
    file = RECOVERY / 'captures' / capture
    raw = file.read_bytes()
    value = json.loads(raw)
    assert not set(value) <= {'i', 'index', 'value', 'result', 'output', 'status', 'chunk_id', 'wall_time_seconds', 'exit_code', 'original_token_count'}
    return save(name, raw, DATA.provenance(META[capture]), 'complete captured JSON observation; original artifact byte hash unavailable', 'Copy verified capture bytes without rerunning the probe.')


def json_payloads(text):
    decoder = json.JSONDecoder()
    for match in re.finditer(r'(?m)^(?=\{|\[)', text):
        try:
            value, length = decoder.raw_decode(text[match.start():])
        except json.JSONDecodeError:
            continue
        if isinstance(value, dict):
            yield value, text[match.start():match.start() + length]


def output_for(capture):
    value = json.loads((RECOVERY / 'captures' / capture).read_text())
    outputs = [(location, node['output']) for location, node in DATA.walk(value) if isinstance(node, dict) and isinstance(node.get('output'), str)]
    assert outputs
    return outputs[-1]
