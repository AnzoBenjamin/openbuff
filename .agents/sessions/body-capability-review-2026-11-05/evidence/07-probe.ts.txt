function fakePeer(options: { callbackBeforeInitialize?: boolean } = {}) {
  const stdoutListeners: Array<(data: Buffer) => void> = []
  const exitListeners: Array<(code: number | null, signal: string | null) => void> = []
  const received: any[] = []
  const docs = new Map<string, { version: number; text: string }>()
  let kills = 0
  let rejectedChanges = 0
  let waitingInitialize: any
  const emitRaw = (data: Buffer) => { for (const listener of stdoutListeners) listener(data) }
  const emit = (message: unknown) => {
    const body = JSON.stringify(message)
    emitRaw(Buffer.from(`Content-Length: ${Buffer.byteLength(body)}\r\n\r\n${body}`))
  }
  const child = {
    stdin: {
      write(chunk: string) {
        const message = JSON.parse(chunk.slice(chunk.indexOf('\r\n\r\n') + 4))
        received.push(message)
        if (options.callbackBeforeInitialize && message.method === 'initialize') {
          waitingInitialize = message
          emit({ jsonrpc: '2.0', id: 'server-config-1', method: 'workspace/configuration', params: { items: [] } })
          return
        }
        if (message.id === 'server-config-1' && waitingInitialize) {
          emit({ jsonrpc: '2.0', id: waitingInitialize.id, result: { capabilities: {} } })
          waitingInitialize = undefined
          return
        }
        if (message.method === 'initialize') emit({ jsonrpc: '2.0', id: message.id, result: { capabilities: {} } })
        if (message.method === 'textDocument/didOpen') {
          const doc = message.params.textDocument
          docs.set(doc.uri, { version: doc.version, text: doc.text })
        }
        if (message.method === 'textDocument/didChange') {
          const doc = message.params.textDocument
          const known = docs.get(doc.uri)
          if (known && doc.version <= known.version) rejectedChanges++
          else docs.set(doc.uri, { version: doc.version, text: message.params.contentChanges[0].text })
        }
        if (message.method === 'textDocument/didClose') docs.delete(message.params.textDocument.uri)
        if (message.method === 'textDocument/hover') emit({ jsonrpc: '2.0', id: message.id, result: { contents: docs.get(message.params.textDocument.uri)?.text ?? '(unopened)' } })
        if (message.method === 'textDocument/definition') emit({ jsonrpc: '2.0', id: message.id, result: null })
      },
      end() {},
    },
    stdout: { on(_event: string, listener: (data: Buffer) => void) { stdoutListeners.push(listener) } },
    on(event: string, listener: any) { if (event === 'exit') exitListeners.push(listener) },
    kill() { kills++; for (const listener of exitListeners) listener(0, 'SIGTERM') },
  }
  return { child, received, docs, emitRaw, kills: () => kills, rejectedChanges: () => rejectedChanges }
}
