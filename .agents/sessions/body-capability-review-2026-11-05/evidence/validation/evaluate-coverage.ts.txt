import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'
import { evaluateAuditCoverage } from '/home/ben/Code/CLI/openbuff/sdk/src/services/audit-intelligence.ts'
import { classifyBreadth, evaluateMinimumShardRule } from '/home/ben/Code/CLI/openbuff/evals/buffbench/plan-sharding-signals.ts'

const base = path.dirname(fileURLToPath(import.meta.url))
const inventory = JSON.parse(fs.readFileSync(`${base}/INVENTORY.json`, 'utf8'))
const receipts = JSON.parse(fs.readFileSync(`${base}/STRUCTURAL-RECEIPTS.json`, 'utf8'))
const featureRecords = JSON.parse(fs.readFileSync(`${base}/FEATURE-CANDIDATES.json`, 'utf8'))
const outOfScope = [
  { id: '.bin', reason: 'Local helper/generated executables; their source and packaging are inspected.' },
  { id: '.commandcode', reason: 'Local integration preferences, not shipped Openbuff product code.' },
  { id: '.vscode', reason: 'Contributor editor settings.' },
  { id: 'agents-graveyard', reason: 'Archived templates outside active workspace runtime; active loading is inspected.' },
]
const native = evaluateAuditCoverage({ inventory, structuralReceipts: receipts, featureRecords, outOfScope })
const breadth = classifyBreadth('Review the codebase on macro and micro feature completeness, CLI SDK agents runtime indexer providers memory security evals and polyglot integration; make a better plan')
const pickerFiles = fs.readdirSync(`${base}/picks`).filter(file => file.endsWith('.json'))
const minimumPairs = evaluateMinimumShardRule({
  breadth,
  signals: { filePickerCount: pickerFiles.length, auditShardCount: receipts.length } as any,
})
const result = {
  snapshotId: inventory.snapshotId,
  nativeAuditCoverageResult: native,
  minimumPairs,
  structuralInspectionCovered: native.invalidReceipts.length === 0 && native.uncoveredSubsystems.length === 0,
  qualification: 'Native complete=false is retained honestly: candidate records are heuristic, and the source/probes show the helper conflates inspection coverage and product readiness. Actual maturity/assurance is recorded separately; no absent/partial feature was promoted to complete.',
  supplementalSourceQualification: 'Native inventory excludes active cli/release sources and hashes large-file prefixes. Reconstructed read ranges and supplemental membership are in REVIEW-VALIDATION.json. CURRENT-FILE-HASHES.json is a fresh current binding; the original 558-file whole-byte map is unavailable. SOURCE-STABILITY.md records historical comparisons and the narrower recovered source-window agreement. Exported structural receipts do not falsely add excluded paths.',
}
fs.writeFileSync(`${base}/COVERAGE-RESULT.json`, JSON.stringify(result, null, 2) + '\n')
console.log(JSON.stringify({ structuralInspectionCovered: result.structuralInspectionCovered, invalidReceipts: native.invalidReceipts, uncoveredSubsystems: native.uncoveredSubsystems, heuristicFeatures: native.incompleteFeatures.length, nativeComplete: native.complete, minimumPairs }, null, 2))
