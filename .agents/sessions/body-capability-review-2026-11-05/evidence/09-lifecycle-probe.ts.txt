const owned = new Map<number, string>()
function state(pid: number): string | null {
  try {
    const cmdline = fs.readFileSync(`/proc/${pid}/cmdline`, 'utf8')
    if (!cmdline.includes(script)) return null
    const stat = fs.readFileSync(`/proc/${pid}/stat`, 'utf8')
    return stat.slice(stat.lastIndexOf(')') + 1).trim().split(/\s+/)[0]
  } catch { return null }
}
async function getPid(file: string): Promise<number> {
  const end = Date.now() + 3000
  while (!fs.existsSync(file) && Date.now() < end) await sleep(10)
  const pid = Number(fs.readFileSync(file, 'utf8'))
  if (!Number.isInteger(pid) || pid <= 1 || !state(pid)) throw new Error('Owned fixture pid was not established.')
  owned.set(pid, file)
  return pid
}
async function cleanup(pid: number) {
  if (state(pid) && owned.has(pid)) { try { process.kill(pid, 'SIGKILL') } catch {} }
  await sleep(30)
}
